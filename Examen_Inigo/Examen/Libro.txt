package Examen;

public class Libro
{

	String titulo;
	String autor;
	String ISBN;
	Boolean prestado;

	public Libro(String titulo, String autor, String iSBN, Boolean prestado)
{

	super();
	this.titulo = titulo;
	this.autor = autor;
	ISBN = iSBN;
	this.prestado = prestado;
}

public String getAutor()
{
	return autor;
}

public String setAutor()
{
	this.autor = autor;
}

public String toString()
{
	return "Libros " + titulo + ", autor " + autor + " ISBN " + ISBN + " prestado " + prestado;
}

}
